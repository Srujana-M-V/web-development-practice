#include <iostream>
using namespace std;

int main() {
    int marks;
    
    // Prompt the user for input
    cout << "Enter your marks: ";
    cin >> marks;
    
    // Check the grade based on marks
    if (marks >= 90 && marks <= 100) {
        cout << "Grade A" << endl; // Marks between 90 and 100
    } else if (marks >= 70 && marks < 90) {
        cout << "Grade B" << endl; // Marks between 70 and 89
    } else if (marks >= 50 && marks < 70) {
        cout << "Grade C" << endl; // Marks between 50 and 69
    } else if (marks >= 30 && marks < 50) {
        cout << "Grade D" << endl; // Marks between 30 and 49
    } else if (marks >= 0 && marks < 30) {
        cout << "Grade F" << endl; // Marks below 30
    } else {
        cout << "Invalid marks" << endl; // For invalid input (marks > 100 or < 0)
    }

    return 0;
}

leap year

#include <iostream>
using namespace std;

int main() {
    int year;
    
    cout << "Enter a year: ";
    cin >> year;  // Ensure you input a valid integer, e.g., 2024
    
    // Check if the year is a leap year
    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        cout << "This year is a leap year." << endl;
    } else {
        cout << "This year is not a leap year." << endl;
    }

    return 0;
}

operators using switch cases

// switch cases on days 
#include <iostream>
//switch cases
using namespace std;
int main(){
    int a,b;
    char x;
    cout<<"enter 2 numbers"<<endl;
    cin>>a>>b;
    cout<<"enter an operator (+,-,%,*) "<<endl;
    cin>>x;
    
    switch(x){
        case '+':
        cout<<"the sum of a and b is"<< a + b <<endl;
        break;
        case '-':
        cout<<"the sub of a and b is"<< a - b <<endl;
        break;
        case '*':
        cout<<"the multiplication of a and b is"<< a * b <<endl;
        break;
        case '/':
            if (b != 0) {
                cout << "The division of a and b is: " << a / b << endl;
            } else {
                cout << "Error: Division by zero is not allowed." << endl;
            }
            break;
        case '%':
            if (b != 0) {
                cout << "The modulus of a and b is: " << a % b << endl;
            } else {
                cout << "Error: Modulus by zero is not allowed." << endl;
            }
            break;
        default:
        cout<<"invalid operator"<<endl;
    }
    return 0;
}



#include <iostream>
using namespace std;

int main() {
    int category, conversionType;
    double value, result;

    // Step 1: Ask user for category
    cout << "Select category:\n";
    cout << "1. Temperature\n2. Length\n3. Weight\n";
    cout << "Enter your choice (1-3): ";
    cin >> category;

    switch (category) {
        // Temperature Conversion
        case 1:
            cout << "Select conversion:\n";
            cout << "1. Celsius to Fahrenheit\n";
            cout << "2. Fahrenheit to Celsius\n";
            cout << "Enter your choice: ";
            cin >> conversionType;

            cout << "Enter value: ";
            cin >> value;

            switch (conversionType) {
                case 1: // Celsius to Fahrenheit
                    result = (value * 9/5) + 32;
                    cout << value << " Celsius = " << result << " Fahrenheit\n";
                    break;
                case 2: // Fahrenheit to Celsius
                    result = (value - 32) * 5/9;
                    cout << value << " Fahrenheit = " << result << " Celsius\n";
                    break;
                default:
                    cout << "Invalid conversion choice!\n";
            }
            break;

        // Length Conversion
        case 2:
            cout << "Select conversion:\n";
            cout << "1. Meters to Kilometers\n";
            cout << "2. Kilometers to Meters\n";
            cout << "Enter your choice: ";
            cin >> conversionType;

            cout << "Enter value: ";
            cin >> value;

            switch (conversionType) {
                case 1: // Meters to Kilometers
                    result = value / 1000;
                    cout << value << " Meters = " << result << " Kilometers\n";
                    break;
                case 2: // Kilometers to Meters
                    result = value * 1000;
                    cout << value << " Kilometers = " << result << " Meters\n";
                    break;
                default:
                    cout << "Invalid conversion choice!\n";
            }
            break;

        // Weight Conversion
        case 3:
            cout << "Select conversion:\n";
            cout << "1. Grams to Kilograms\n";
            cout << "2. Kilograms to Grams\n";
            cout << "Enter your choice: ";
            cin >> conversionType;

            cout << "Enter value: ";
            cin >> value;

            switch (conversionType) {
                case 1: // Grams to Kilograms
                    result = value / 1000;
                    cout << value << " Grams = " << result << " Kilograms\n";
                    break;
                case 2: // Kilograms to Grams
                    result = value * 1000;
                    cout << value << " Kilograms = " << result << " Grams\n";
                    break;
                default:
                    cout << "Invalid conversion choice!\n";
            }
            break;

        default:
            cout << "Invalid category choice!\n";
    }

    return 0;
}

WHILE STATEMENTS


#include <iostream>
using namespace std;

int main() {
    int a = 1; // Start at 1

    while (a <= 30) { // Loop from 1 to 20
        if (a % 3 == 0) { // Check if a is even
            cout << a << endl; // Print the even number
        }
        a++; // Increment to the next number
    }

    return 0; // End program
}
