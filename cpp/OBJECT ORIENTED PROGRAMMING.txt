#include<iostream>
using namespace std;
class emp{
    private:
    string name;
    float salary;
    int id;
    int yoe;
    public:
    void setdata(string n,float s,int i,int y){
        name=n;
        salary=s;
        id=i;
        yoe=y;
    }
    void getdata(){
        cout<<"name:"<<name<<endl;
        cout<<"salary:"<<salary<<endl;
        cout<<"id:"<<id<<endl;
        cout<<"yoe:"<<yoe<<endl;
    }
    void applybonus() {
        if (yoe > 5) {
            salary += salary * 0.20;
        } else if (yoe > 2) {
            salary += salary * 0.10;
        } else {
            // no bonus
        }
    }
};
int main(){
    emp e;
    e.setdata("srujana",50000,158,3);
    e.getdata();
    e.applybonus();
    e.getdata();
    return 0;
}

Even if a function changes the data (like applybonus()), it does not automatically print anything.
To see the updated data, you must call getdata() after calling such a function.
applybonus() just changes the internal value (like salary).
It does not print anything by itself (because it's void and has no cout).
To see the effect, you must call a function like getdata() that prints the updated values.

#include<iostream>
using namespace std;

class bank {
private:
    string name;
    int accno;
    float balance;

public:
    void openacc() {
        cout << "Enter your name: ";
        cin >> name;
        cout << "Enter your account number: ";
        cin >> accno;
        cout << "Enter initial balance: ";
        cin >> balance;
    }

    void deposit() {
        float amount;
        cout << "Enter amount to deposit: ";
        cin >> amount;
        balance += amount;
        cout << "Deposited: " << amount << endl;
    }

    void withdraw() {
        float amount;
        cout << "Enter amount to withdraw: ";
        cin >> amount;
        if (balance >= amount) {
            balance -= amount;
            cout << "Withdrawn: " << amount << endl;
        } else {
            cout << "Insufficient balance!" << endl;
        }
    }

    void display() {
        cout << "\n--- Account Details ---" << endl;
        cout << "Name: " << name << endl;
        cout << "Account Number: " << accno << endl;
        cout << "Balance: " << balance << endl;
        cout << "-----------------------\n" << endl;
    }
};

int main() {
    bank b;
    int choice;

    b.openacc();

    do {
        cout << "\nMenu:\n1. Deposit\n2. Withdraw\n3. Display\n4. Exit" << endl;
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice) {
        case 1:
            b.deposit();
            break;
        case 2:
            b.withdraw();
            break;
        case 3:
            b.display();
            break;
        case 4:
            cout << "Exiting program..." << endl;
            break;
        default:
            cout << "Invalid choice!" << endl;
        }

    } while (choice != 4);

    return 0;
}
