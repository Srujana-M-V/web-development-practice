CALL BY VALUE AND REFERENCE

#include<iostream>
using namespace std;

void swap(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}

int main() {
    int a, b;
    cout << "Enter a and b: ";
    cin >> a >> b;

    swap(a, b);

    cout << "After swapping:" << endl;
    cout << "a = " << a << endl;
    cout << "b = " << b << endl;

    return 0;
}


#include<iostream>  
using namespace std;

// 📦 This function swaps three variables in a specific pattern:
// a = b, b = c, c = a (using temp)
void swap(int &a, int &b, int &c) {
    int temp = a;  // 🔁 Store original 'a' temporarily
    a = b;         // 🔁 Now 'a' becomes value of 'b'
    b = c;         // 🔁 'b' becomes value of 'c'
    c = temp;      // 🔁 'c' gets the original 'a' from temp
}

int main() {
    int a, b, c;
    cout << "enter a and b and c: ";
    cin >> a >> b >> c;     // 🧠 User inputs the values

    swap(a, b, c);          // ✨ Calling the function - changes reflect here because of reference

    cout << a << " " << b << " " << c;  // ✅ Final output after swapping
    return 0;
}

INLINE FUNCTIONS

An inline function is a function where the compiler replaces the function call with the actual code of the function. It helps to reduce the overhead of a function call, especially for small, frequently used functions.

📌 Why use Inline?
It’s faster because no jumping to function code and back (no function call overhead).

Mostly used for small functions.


#include<iostream>
using namespace std;
inline int max(int x,int y,int z){
    if(x>y&&x>z){
        cout<<"x is greater";
        return x;
    }
    else if(y>z){
        cout<<"y is greater";
        return y;
    }
    else{
        cout<<"z is greater";
        return z;
    }
}
int main(){
    int a,b,c;
    cout<<" enter value of a,b,c: ";
    cin>>a>>b>>c;
    cout<<"\nmax num is"<<max(a,b,c);
    return 0;
}

RECURSION AND FUNCTIONS

#include<iostream>              // Preprocessor directive to include input-output stream
using namespace std;            // To avoid writing std:: before cin and cout

// --- Function Declaration + Definition ---
int fact(int n) {               // Function to calculate factorial of n
    if(n == 1 || n == 0) {      // Base condition: if n is 1 or 0, return 1
        return 1;               // This is the stopping condition for recursion
    }
    else {
        return n * fact(n - 1); // Recursive call: fact(n) = n * fact(n-1)
    }
}

fact(4) → 4 * fact(3)
       → 4 * (3 * fact(2))
       → 4 * (3 * (2 * fact(1)))
       → 4 * (3 * (2 * 1)) → 24

#include<iostream>
using namespace std;

int area(int length) {
    return length * length;
}

int area(int length, int breadth) {
    return length * breadth;
}

float areaCircle(float radius) {  // renamed to avoid ambiguity
    return 3.14 * radius * radius;
}

int main() {
    int length, breadth;
    float radius;

    cout << "Enter length and breadth: ";
    cin >> length >> breadth;

    cout << "Enter radius: ";
    cin >> radius;

    cout << "Area of square: " << area(length) << endl;
    cout << "Area of rectangle: " << area(length, breadth) << endl;
    cout << "Area of circle: " << areaCircle(radius) << endl;

    return 0;
}


#include<iostream>
using namespace std;

// Function to find the maximum of two integers
int max(int a, int b) {
    if (a > b) {
        return a;
    } else {
        return b;
    }
}

// Function to find the maximum of three integers
int max(int a, int b, int c) {
    if (a > b && a > c) {
        return a;
    } else if (b > c) {
        return b;
    } else {
        return c;
    }
}

// Function to find the maximum of two floats
float max(float x, float y) {
    if (x > y) {
        return x;
    } else {
        return y;
    }
}

int main() {
    int a, b, c;
    float x, y;
    
    // Taking input for the numbers
    cout << "Enter two integers (a and b): ";
    cin >> a >> b;
    
    cout << "Enter three integers (a, b, and c): ";
    cin >> a >> b >> c;
    
    cout << "Enter two floats (x and y): ";
    cin >> x >> y;
    
    // Output the maximum values
    cout << "Maximum of two integers: " << max(a, b) << endl;
    cout << "Maximum of three integers: " << max(a, b, c) << endl;
    cout << "Maximum of two floats: " << max(x, y) << endl;

    return 0;
}
